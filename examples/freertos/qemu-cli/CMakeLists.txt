cmake_minimum_required(VERSION 3.20)
# Runs on qemu mps2-an385 (Cortex-M3). Build + run: scripts/app run qemu-cli   (needs: scripts/setup install core)
if(NOT DEFINED ENV{FREERTOS_QEMU})
  message(FATAL_ERROR "source toolchain/freertos/env.sh first (scripts/app does it)")
endif()
set(CMAKE_TOOLCHAIN_FILE $ENV{FREERTOS_QEMU}/arm-none-eabi.cmake)
set(BOARD "mps2-an385" CACHE STRING "qemu mps2-an385 only")
project(qemu-cli C ASM)
include($ENV{FREERTOS_QEMU}/freertos_qemu.cmake)
add_executable(qemu-cli src/main.c)
option(CLI_INTERACTIVE "keep a UART shell running after the self-test" OFF)
freertos_qemu(qemu-cli CLI DEFINES $<$<BOOL:${CLI_INTERACTIVE}>:CLI_INTERACTIVE>)
