cmake_minimum_required(VERSION 3.20)
# @NAME@: the same app for qemu (BOARD=mps2-an385, default) or an STM32 board (BOARD=blackpill-f411, ...).
if(NOT DEFINED ENV{FREERTOS_KERNEL})
  message(FATAL_ERROR "source <freertos-box>/toolchain/freertos/env.sh first (scripts/app does it)")
endif()
set(BOARD "mps2-an385" CACHE STRING "mps2-an385 (qemu) | blackpill-f411 | weact-g431 | mini-h743 | bluepill-f103")
if(BOARD STREQUAL "mps2-an385")
  set(CMAKE_TOOLCHAIN_FILE $ENV{FREERTOS_QEMU}/arm-none-eabi.cmake)
  project(@NAME@ C ASM)
  include($ENV{FREERTOS_QEMU}/freertos_qemu.cmake)
  add_executable(@NAME@ src/app.c platform/qemu.c)
  freertos_qemu(@NAME@ INCLUDES ${CMAKE_CURRENT_SOURCE_DIR}/platform)
else()
  set(STM32_BOARDS "$ENV{STM32_CUBE_F4}/../boards" CACHE PATH "toolchain/stm32/boards")
  set(CMAKE_TOOLCHAIN_FILE ${STM32_BOARDS}/arm-none-eabi.cmake)
  project(@NAME@ C ASM)
  include(${STM32_BOARDS}/board.cmake)
  include($ENV{FREERTOS_KERNEL}/../common/freertos.cmake)
  add_executable(@NAME@ src/app.c platform/stm32.c)
  target_include_directories(@NAME@ PRIVATE platform)
  freertos_board(@NAME@)
endif()
